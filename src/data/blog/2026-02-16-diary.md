## パスワード付きPDFのパスワードを除去するスクリプトをVibe Codingした

給与明細を調べたくなったので、調べたところ、給与明細にパスワードがかかっていた。

閲覧するたびパスワードを入れるのは大変なので、除去する。

### memo.md

```markdown
このディレクトリにはパスワード保護されたPDFがあります。

これらは私の給与明細書です。

パスワード保護を外して、誰でも閲覧できる状態にしたいです。

パスワード保護を外したPDFファイルを生成するプログラムを書いてください。
```

`claude "@memo.md"` してしばらく待機

### unlock-pdfs.py

`uv run unlock-pdfs.py` して実行する形式

```python
#!/usr/bin/env python3
"""Remove password protection from PDF files in a directory.

This script reads password-protected PDF files from the current directory
and writes unlocked copies to an 'unlocked/' subdirectory.

Usage:
    uv run unlock_pdfs.py
"""

import sys
from dataclasses import dataclass, field
from pathlib import Path

import pikepdf

PASSWORD_FILE = ".pdf_password"


@dataclass
class ProcessingResult:
    """Result of processing a single PDF file."""

    path: Path
    success: bool
    error: str = ""


@dataclass
class Summary:
    """Aggregated results of processing all PDF files."""

    results: list[ProcessingResult] = field(default_factory=list)

    @property
    def succeeded(self) -> list[ProcessingResult]:
        """Return results where processing succeeded."""
        return [r for r in self.results if r.success]

    @property
    def failed(self) -> list[ProcessingResult]:
        """Return results where processing failed."""
        return [r for r in self.results if not r.success]


def read_password(password_file: Path) -> str:
    """Read the PDF password from a plain-text file.

    The file should contain only the password on the first line.
    Leading and trailing whitespace (including newlines) are stripped.

    Args:
        password_file: Path to the file containing the password.

    Returns:
        The password string read from the file.

    Raises:
        SystemExit: If the file does not exist or cannot be read.

    Example:
        >>> # Given a file .pdf_password containing "s3cr3t"
        >>> read_password(Path(".pdf_password"))
        's3cr3t'
    """
    try:
        return password_file.read_text(encoding="utf-8").splitlines()[0].strip()
    except FileNotFoundError:
        print(f"Error: Password file not found: {password_file}", file=sys.stderr)
        print(f"  Create it with: echo 'your_password' > {password_file}", file=sys.stderr)
        sys.exit(1)
    except (OSError, IndexError) as e:
        print(f"Error: Could not read password file {password_file}: {e}", file=sys.stderr)
        sys.exit(1)


def unlock_pdf(source: Path, output: Path, password: str) -> ProcessingResult:
    """Remove password protection from a single PDF file.

    Opens the source PDF with the given password and saves an unlocked
    copy to the output path. The source file is never modified.

    Args:
        source: Path to the password-protected PDF.
        output: Path where the unlocked PDF will be written.
        password: Password used to open the PDF.

    Returns:
        A ProcessingResult indicating success or failure with an error message.

    Example:
        >>> result = unlock_pdf(Path("locked.pdf"), Path("unlocked/locked.pdf"), "secret")
        >>> result.success
        True
    """
    try:
        with pikepdf.open(source, password=password) as pdf:
            pdf.save(output)
        return ProcessingResult(path=source, success=True)
    except pikepdf.PasswordError:
        return ProcessingResult(
            path=source,
            success=False,
            error="Incorrect password",
        )
    except pikepdf.PdfError as e:
        return ProcessingResult(
            path=source,
            success=False,
            error=f"PDF error: {e}",
        )
    except OSError as e:
        return ProcessingResult(
            path=source,
            success=False,
            error=f"I/O error: {e}",
        )


def process_directory(src: Path, out: Path, password: str) -> Summary:
    """Process all PDF files in a directory, saving unlocked copies.

    Finds every *.pdf file in src (case-insensitive), unlocks it using
    the provided password, and saves the result to the out directory,
    preserving the original filename.

    Args:
        src: Directory containing the password-protected PDF files.
        out: Directory where unlocked PDFs will be written (created if absent).
        password: Password used to open each PDF.

    Returns:
        A Summary containing individual ProcessingResult for each file.
    """
    out.mkdir(parents=True, exist_ok=True)

    pdf_files = sorted(src.glob("*.pdf")) + sorted(src.glob("*.PDF"))
    # Deduplicate in case of mixed case on case-sensitive filesystems
    seen: set[Path] = set()
    unique_pdfs: list[Path] = []
    for p in pdf_files:
        if p not in seen:
            seen.add(p)
            unique_pdfs.append(p)

    summary = Summary()
    total = len(unique_pdfs)

    for i, pdf_path in enumerate(unique_pdfs, start=1):
        dest = out / pdf_path.name
        print(f"[{i}/{total}] {pdf_path.name} ... ", end="", flush=True)
        result = unlock_pdf(pdf_path, dest, password)
        if result.success:
            print("OK")
        else:
            print(f"FAILED ({result.error})")
        summary.results.append(result)

    return summary


def print_summary(summary: Summary) -> None:
    """Print a human-readable summary of processing results.

    Args:
        summary: The Summary object containing all processing results.
    """
    total = len(summary.results)
    ok = len(summary.succeeded)
    ng = len(summary.failed)

    print()
    print("=" * 50)
    print(f"Summary: {ok}/{total} succeeded, {ng} failed")

    if summary.failed:
        print()
        print("Failed files:")
        for r in summary.failed:
            print(f"  - {r.path.name}: {r.error}")

    print("=" * 50)


def main() -> int:
    """Entry point for the PDF unlock script.

    Reads the password from PASSWORD_FILE in the current directory,
    then processes all PDFs and writes unlocked copies to 'unlocked/'.

    Returns:
        0 if all files were processed successfully, 1 otherwise.
    """
    src = Path.cwd()
    out = src / "unlocked"

    pdf_count = len(list(src.glob("*.pdf"))) + len(list(src.glob("*.PDF")))
    if pdf_count == 0:
        print(f"No PDF files found in {src}", file=sys.stderr)
        return 1

    print(f"Found {pdf_count} PDF file(s) in {src}")
    print(f"Unlocked copies will be saved to: {out}")
    print()

    password_file = src / PASSWORD_FILE
    password = read_password(password_file)
    print(f"Password loaded from: {password_file}")
    print()

    summary = process_directory(src, out, password)
    print_summary(summary)

    return 0 if not summary.failed else 1


if __name__ == "__main__":
    sys.exit(main())
```
